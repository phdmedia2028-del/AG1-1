===== FILE: firestore.rules =====
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isRootAdmin() {
      return request.auth != null
        && (request.auth.token.email == 'ntunga.26@gmail.com'
            || request.auth.token.email == 'tunganhhh.nl@gmail.com');
    }

    function isAdmin() {
      return request.auth != null && (
        isRootAdmin()
        || exists(/databases/$(database)/documents/admins/$(request.auth.token.email))
      );
    }

    // Public waitlist: anyone can read and submit; only admins can modify or delete
    match /waitlist/{docId} {
      allow read: if true;
      allow create: if true;
      allow update, delete: if isAdmin();
    }

    // Slot counter: public read, admin write
    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Admin allowlist: only the root admin manages it
    match /admins/{email} {
      allow read: if isAdmin();
      allow create, update, delete: if isRootAdmin();
    }
  }
}

/** Nối mảng theo lối tiếng Việt: a, b và c */
function joinVi(arr: string[]): string {
  if (arr.length <= 1) return arr[0] || "";
  return arr.slice(0, -1).join(", ") + " và " + arr[arr.length - 1];
}

/** Bỏ dấu tiếng Việt, bỏ ký tự đặc biệt và khoảng trắng */
function slugify(input: string): string {
  return input
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/đ/g, "d").replace(/Đ/g, "D")
    .replace(/[^a-zA-Z0-9]/g, "");
}

const onlyDigits = (s: string) => s.replace(/\D/g, "");

const formatPrice = (n: number) => n.toLocaleString("vi-VN") + "đ";

/** Nội dung chuyển khoản = tiền tố + SĐT + tên Discord, tối đa 50 ký tự */
function buildTransferContent(phone: string, discord: string): string {
  const p = onlyDigits(phone);
  const d = slugify(discord);
  if (!p && !d) return "SDT TENDISCORD";
  const raw = `${CONFIG.transferPrefix}${p} ${d}`.trim();
  return raw.slice(0, MAX_CONTENT_LEN);
}

/** Link QR VietQR tự sinh, đã nhúng số tiền và nội dung */
function buildQrUrl(content: string): string {
  if (CONFIG.qrImageUrl) return CONFIG.qrImageUrl;
  const base = `https://img.vietqr.io/image/${CONFIG.bankId}-${CONFIG.accountNo}-compact2.png`;
  const params = new URLSearchParams({
    amount: String(CONFIG.price),
    addInfo: content,
    accountName: CONFIG.accountName,
  });
  return `${base}?${params.toString()}`;
}

/**
 * Nén ảnh bill phía trình duyệt trước khi gửi.
 * Hạ chiều rộng tối đa 1200px, xuất JPEG chất lượng 0.8.
 * Giúp payload từ vài MB xuống còn khoảng 200-400KB.
 */
async function compressImage(
  file: File
): Promise<{ base64: string; type: string; name: string }> {
  const dataUrl = await new Promise<string>((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(reader.result as string);
    reader.onerror = reject;
    reader.readAsDataURL(file);
  });

  const img = await new Promise<HTMLImageElement>((resolve, reject) => {
    const image = new Image();
    image.onload = () => resolve(image);
    image.onerror = reject;
    image.src = dataUrl;
  });

  const MAX_W = 1200;
  const scale = Math.min(1, MAX_W / img.width);
  const w = Math.round(img.width * scale);
  const h = Math.round(img.height * scale);

  const canvas = document.createElement("canvas");
  canvas.width = w;
  canvas.height = h;
  const ctx = canvas.getContext("2d");
  if (!ctx) {
    // Không dùng được canvas thì gửi nguyên bản
    return {
      base64: dataUrl.split(",")[1] || "",
      type: file.type,
      name: file.name,
    };
  }
  ctx.fillStyle = "#FFFFFF";
  ctx.fillRect(0, 0, w, h);
  ctx.drawImage(img, 0, 0, w, h);

  const out = canvas.toDataURL("image/jpeg", 0.8);
  const baseName = file.name.replace(/\.[^.]+$/, "");
  return {
    base64: out.split(",")[1] || "",
    type: "image/jpeg",
    name: `${baseName}.jpg`,
  };
}

function CopyButton({ value, label }: { value: string; label: string }) {
  const [copied, setCopied] = useState(false);
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);

  const handleCopy = async () => {
    try {
      await navigator.clipboard.writeText(value);
    } catch {
      // Trình duyệt cũ hoặc không có quyền clipboard
      const ta = document.createElement("textarea");
      ta.value = value;
      document.body.appendChild(ta);
      ta.select();
      try { document.execCommand("copy"); } catch { /* bỏ qua */ }
      document.body.removeChild(ta);
    }
    setCopied(true);
    if (timer.current) clearTimeout(timer.current);
    timer.current = setTimeout(() => setCopied(false), 1500);
  };

  useEffect(() => () => { if (timer.current) clearTimeout(timer.current); }, []);

  return (
    <button
      type="button"
      onClick={handleCopy}
      aria-label={`Sao chép ${label}`}
      className="shrink-0 w-7 h-7 rounded-lg bg-[#C9A961]/15 border border-[#C9A961]/30 flex items-center justify-center text-[#C9A961] hover:bg-[#C9A961]/25 transition-colors cursor-pointer"
    >
      {copied ? <Check className="w-3.5 h-3.5" /> : <Copy className="w-3.5 h-3.5" />}
    </button>
  );
}

  "w-full bg-white/5 border border-white/10 rounded-xl px-4 py-3 text-sm text-[#F5F4F0] placeholder:text-slate-600 focus:outline-none focus:border-[#C9A961]/60 focus:ring-2 focus:ring-[#C9A961]/15 transition-all";

export default function WaitlistModal({ open, onClose, onSubmitted }: Props) {
  const [name, setName] = useState("");
  const [zalo, setZalo] = useState("");
  const [error, setError] = useState("");
  const [submitting, setSubmitting] = useState(false);
  const [result, setResult] = useState<{ position: number; duplicate: boolean } | null>(null);

  if (!open) return null;

  const resetAndClose = () => {
    if (submitting) return;
    setName(""); setZalo(""); setError(""); setResult(null);
    onClose();
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError("");
    const cleanName = name.trim().replace(/\s+/g, " ");
    const digits = zalo.replace(/\D/g, "");
    if (!cleanName) { setError("Vui lòng nhập họ và tên."); return; }
    if (digits.length < 9 || digits.length > 12) { setError("Số Zalo chưa hợp lệ. Vui lòng kiểm tra lại."); return; }

    setSubmitting(true);
    try {
      const r = await addToWaitlist(cleanName, digits);
      setResult(r);
      onSubmitted?.();
    } catch (err) {
      const msg = err instanceof Error ? err.message : "";
      // Mã lỗi giúp admin biết nguyên nhân; khách chỉ cần thử lại hoặc nhắn Zalo
      const code =
        err instanceof TypeError ? "KHÔNG KẾT NỐI ĐƯỢC MÁY CHỦ"
        : /Thiếu thông tin bắt buộc/.test(msg) ? "APPS SCRIPT CHƯA CẬP NHẬT CODE HÀNG CHỜ"
        : /permission|quyền|Exception/i.test(msg) ? "APPS SCRIPT CHƯA CÓ QUYỀN SỬA SHEET HÀNG CHỜ"
        : msg || "KHÔNG RÕ";
      setError(`Chưa gửi được đăng ký. Vui lòng thử lại sau ít phút hoặc nhắn Zalo để được hỗ trợ. (Mã lỗi: ${code})`);
    }
    setSubmitting(false);
  };

  return (
    <div
      className="fixed inset-0 z-[10000] flex items-end sm:items-center justify-center sm:p-4 bg-black/70 backdrop-blur-sm"
      onClick={resetAndClose}
    >
      <div
        className="relative w-full sm:max-w-md bg-[#14130F] border border-[#C9A961]/30 rounded-t-3xl sm:rounded-3xl shadow-2xl max-h-[92vh] overflow-y-auto"

      >
        <button
          onClick={resetAndClose}
          className="absolute top-4 right-4 w-9 h-9 rounded-full bg-white/5 border border-[#C9A961]/20 flex items-center justify-center text-slate-400 hover:text-[#C9A961] hover:border-[#C9A961]/50 transition-all cursor-pointer z-10"
          aria-label="Đóng"
        >
          <X className="w-4 h-4" />
        </button>

        {result ? (
          <div className="p-7 sm:p-9 text-center space-y-6">
            <div className="w-16 h-16 rounded-full bg-[#C9A961]/15 border border-[#C9A961]/40 flex items-center justify-center mx-auto">
              <Check className="w-8 h-8 text-[#C9A961]" />
            </div>
            <div className="space-y-2">
              <h3 className="font-display font-extrabold text-2xl text-[#F5F4F0]">
                {result.duplicate ? "Bạn đã có trong hàng chờ" : "Bạn đã vào hàng chờ"}
              </h3>
              {result.position > 0 && (
                <>
                  <p className="text-slate-400 text-sm">Vị trí của bạn trong danh sách</p>

                </>
              )}
            </div>
            <div className="bg-white/5 border border-[#C9A961]/20 rounded-2xl p-5 text-left">

                Khi slot coaching mở lại, Hiếu sẽ liên hệ với bạn qua Zalo theo thứ tự trong hàng chờ. Hãy để ý tin nhắn Zalo của bạn nhé.
              </p>
            </div>
            <a
              href={ZALO_LINK}
              target="_blank"
              rel="noopener noreferrer"
              className="w-full inline-flex items-center justify-center px-6 py-3.5 rounded-xl text-sm font-bold bg-white/5 border border-[#C9A961]/30 text-[#C9A961] hover:bg-[#C9A961]/10 transition-all cursor-pointer"

// Hàng chờ Coaching lưu trên Google Sheet "Danh Sách Chờ Coaching",
// đọc/ghi qua cùng Apps Script Web App với phần thanh toán.
export const WAITLIST_API =
  "https://script.google.com/macros/s/AKfycbxrOh1Fo7u5c75pX6t4eAOtPFGEhxjUToLkvQWAdivXxXOIa2InRqHoalffaVW3PUh0bA/exec";

export interface WaitlistData {
  count: number;
  names: string[];
}

const WAITLIST_SHEET_ID = "1JxiGauAd-2fj7AS7TSOffJDTcKbSjkrMUlGGcNSt8UY";

/** Cách 2 (dự phòng): qua Apps Script (chỉ trả tên). */
async function fromScript(): Promise<WaitlistData> {
  const res = await fetch(`${WAITLIST_API}?action=waitlist&t=${Date.now()}`);
  const json = await res.json();
  if (!json || !json.ok || !Array.isArray(json.names)) throw new Error(json?.error || "Script chưa trả danh sách");
  return { count: json.names.length, names: json.names };
}

/** Cách 1 (chính): đọc thẳng cột B của sheet bằng JSONP (không bị chặn CORS).
 *  Sheet cần để "Bất kỳ ai có đường liên kết đều xem được". */
function fromSheet(): Promise<WaitlistData> {
  return new Promise((resolve, reject) => {
    const cb = `__ag1Waitlist${Date.now()}${Math.floor(Math.random() * 1000)}`;
    const script = document.createElement("script");
    const cleanup = () => {
      try { delete (window as any)[cb]; } catch { (window as any)[cb] = undefined; }
      script.remove();
      clearTimeout(timer);
    };
    const timer = setTimeout(() => { cleanup(); reject(new Error("Hết thời gian đọc sheet")); }, 10000);
    (window as any)[cb] = (resp: any) => {
      cleanup();
      if (!resp || resp.status === "error" || !resp.table) { reject(new Error("Sheet trả lỗi")); return; }
      const names: string[] = (resp.table.rows || [])
        .map((r: any) => String(r?.c?.[0]?.v ?? "").trim())
        .filter(Boolean);
      resolve({ count: names.length, names });
    };
    script.onerror = () => { cleanup(); reject(new Error("Không đọc được sheet")); };
    script.src =
      `https://docs.google.com/spreadsheets/d/${WAITLIST_SHEET_ID}/gviz/tq` +
      `?tqx=out:json;responseHandler:${cb}&range=B2:B1000&headers=0&t=${Date.now()}`;
    document.head.appendChild(script);
  });
}

/** Lấy danh sách tên đang chờ: đọc thẳng sheet trước, lỗi thì hỏi Apps Script. */
export async function fetchWaitlist(): Promise<WaitlistData> {
  try {
    return await fromSheet();
  } catch {
    return await fromScript();
  }
}

/** Ghi một người vào hàng chờ. Trả về vị trí trong danh sách. */
export async function addToWaitlist(name: string, zalo: string): Promise<{ position: number; duplicate: boolean }> {
  const res = await fetch(WAITLIST_API, {
    method: "POST",
    headers: { "Content-Type": "text/plain;charset=utf-8" },
    body: JSON.stringify({ action: "waitlist", name, zalo }),
  });
  const json = await res.json();
  if (!json || !json.ok) throw new Error(json?.error || "Không gửi được đăng ký");
  return { position: Number(json.position) || 0, duplicate: !!json.duplicate };
}

/** Che số điện thoại, chỉ để lộ 4 số cuối. VD: 0812338344 -> ******8344 */
export function maskPhone(phone: string): string {
  const digits = phone.replace(/\D/g, "");
  if (digits.length <= 4) return digits;
  return "*".repeat(digits.length - 4) + digits.slice(-4);
}

/** Chuẩn hoá số điện thoại để so sánh trùng */
export function normalizePhone(phone: string): string {
  return phone.replace(/\D/g, "");
}

/** "2026-11" -> "Tháng 11/2026" */
export function formatStartMonth(value: string): string {
  if (!value || !value.includes("-")) return value;
  const [year, month] = value.split("-");
  return `Tháng ${parseInt(month, 10)}/${year}`;
}
